```java
package com.example.post.dto;

import java.time.LocalDateTime;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;

    private Long postNo;

    private LocalDateTime regDate;
    private LocalDateTime updDate;

    public Long getPostNo() { return postNo; }

    public void setTitle(String title) { this.title = title; }
    public String getWriter() { return writer; }
    public void setWriter(String writer) { this.writer = writer; }
    public String getContent() { return content; }
    public void setContent(String content) { this.content = content; }
```

```java
package com.example.post.service;

import java.util.List;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;
import com.example.post.dto.Page;
import com.example.post.dto.Post;
import com.example.post.mapper.PostMapper;

@Service
@Transactional(readOnly = true)
public class PostServiceImpl implements PostService {
    private final PostMapper postMapper;
    public PostServiceImpl(PostMapper postMapper) { this.postMapper = postMapper; }
    @Override
    public List<Post> list(Page page) { return postMapper.list(page); }
    @Override
    public long count() { return postMapper.count(); }
    @Override
    public Post read(Long postNo) {
        Post post = postMapper.read(postNo);
        if (post == null) throw notFound();
        return post;
    }
    @Override
    @Transactional
    public void insert(Post post) {
        // 등록 요청에 글번호가 있어도 DB가 생성한 번호를 사용한다.
        post.setPostNo(null);
        postMapper.insert(post);
    }
    @Override
    @Transactional
    public void update(Post post) {
        read(post.getPostNo());
        postMapper.update(post);
    }
    @Override
    @Transactional
    public void delete(Long postNo) {
        if (postMapper.delete(postNo) == 0) throw notFound();
    }
    private ResponseStatusException notFound() {
        return new ResponseStatusException(HttpStatus.NOT_FOUND, "게시글이 없습니다.");
    }
}
```

```java
package com.example.post.controller;

import jakarta.validation.Valid;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.validation.BindingResult;
import org.springframework.web.bind.annotation.*;
import com.example.post.dto.Page;
import com.example.post.dto.Post;
import com.example.post.service.PostService;

@Controller
public class PostController {
    private final PostService postService;
    public PostController(PostService postService) { this.postService = postService; }

    @GetMapping("/")
    public String index() { return "index"; }

    @GetMapping("/Post/list")
    public String list(@RequestParam(name = "page", defaultValue = "1") int page,
                       Model model) {
        Page pageInfo = new Page(page, postService.count());
        model.addAttribute("postList", postService.list(pageInfo));
        model.addAttribute("pageInfo", pageInfo);
        return "Post/list";
    }

    @GetMapping("/Post/read")
    public String read(@RequestParam("PostNo") Long postNo, Model model) {
        model.addAttribute("post", postService.read(postNo));
        return "Post/read";
    }

    @GetMapping("/Post/insert")
    public String insertForm(Model model) {
        model.addAttribute("post", new Post());
        return "Post/insert";
    }

    @PostMapping("/Post/insert")
    public String insert(@Valid @ModelAttribute("post") Post post,
                         BindingResult result) {
        if (result.hasErrors()) return "Post/insert";
        postService.insert(post);
        return "redirect:/Post/read?PostNo=" + post.getPostNo();
    }

    @GetMapping("/Post/update")
    public String updateForm(@RequestParam("PostNo") Long postNo, Model model) {
        model.addAttribute("post", postService.read(postNo));
        return "Post/update";
    }

    @PostMapping("/Post/update")
    public String update(@RequestParam("PostNo") Long postNo,
                         @Valid @ModelAttribute("post") Post post,
                         BindingResult result) {
        post.setPostNo(postNo);
        if (result.hasErrors()) return "Post/update";
        postService.update(post);
        return "redirect:/Post/read?PostNo=" + postNo;
    }

    @PostMapping("/Post/delete")
    public String delete(@RequestParam("PostNo") Long postNo) {
        postService.delete(postNo);
        return "redirect:/Post/list";
    }
}
```

## 문항5 · 페이징 최종 코드

추가: Page.java, Service/Mapper의 count(). 변경: list(Page), Controller의 page 파라미터와 pageInfo 전달, XML의 LIMIT/OFFSET. 목록 뷰에도 페이지 이동 링크를 추가했습니다. Post.java는 문항2 코드 그대로입니다.

### src/main/java/com/example/post/dto/Page.java

```java
package com.example.post.dto;

/** 한 페이지 10개, 페이지 번호 묶음 5개. 빈 목록도 1페이지로 표시한다. */
public class Page {
    private final int page;
    private final int size = 10;
    private final int blockSize = 5;
    private final long totalCount;
    private final int totalPages;
    private final int startPage;
    private final int endPage;

    public Page(int requestedPage, long totalCount) {
        this.totalCount = Math.max(0L, totalCount);
        this.totalPages = (int) Math.max(1L, (this.totalCount + size - 1) / size);
        this.page = Math.min(Math.max(1, requestedPage), totalPages);
        this.startPage = ((page - 1) / blockSize) * blockSize + 1;
        this.endPage = Math.min(startPage + blockSize - 1, totalPages);
    }
    public int getPage() { return page; }
    public int getSize() { return size; }
    public long getTotalCount() { return totalCount; }
    public int getTotalPages() { return totalPages; }
    public int getStartPage() { return startPage; }
    public int getEndPage() { return endPage; }
    public long getOffset() { return (long) (page - 1) * size; }
    public boolean isHasPrevious() { return page > 1; }
    public boolean isHasNext() { return page < totalPages; }
}
```
